# psycopg2 is the PostgreSQL driver. Only src/database.py and the database

# lazy, so stages 3, 4 and --dry-run all work without it installed.
#
# psycopg2-binary ships prebuilt wheels, so no compiler or libpq headers are
# required. Switch to the source distribution (psycopg2==2.9.12) if you need a
# build against your own libpq.
psycopg2-binary==2.9.12
